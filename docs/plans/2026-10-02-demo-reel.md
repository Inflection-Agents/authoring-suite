```python
#!/usr/bin/env python3
"""Check a demo-reel shot list against its voiceover script before anything is captured.

A shot list is wrong long before a frame renders: a cue the narration lands on with nothing to show,
a shot filed under a scene that was renamed, a cut that runs twice its target, or a stubbed
dependency shown without saying so. Each of those is cheap to fix here and expensive after capture.

Usage:
    python3 verify_shots.py demo/shots.yaml demo/script.md
"""
from __future__ import annotations

import re
import sys
from pathlib import Path

CUE_RE = re.compile(r"\[cue:([a-z0-9-]+)\]")
SCENE_RE = re.compile(r"^##\s+(?P<id>[A-Za-z]+\d+)\b")
TOLERANCE = 0.10


def script_cues(text: str) -> dict[str, list[str]]:
    """Cue ids per scene, in order, ignoring direction notes (lines starting with '>')."""
    out: dict[str, list[str]] = {}
    scene = None
    for line in text.splitlines():
        m = SCENE_RE.match(line)
        if m:
            scene = m.group("id")
            out[scene] = []
            continue
        if scene is None or line.lstrip().startswith(">"):
            continue
        out[scene].extend(CUE_RE.findall(line))
    return out


def _stubs(slots) -> list[str]:
    found = []
    if isinstance(slots, dict):
        if "stub" in slots:
            found.append(str(slots["stub"]))
        for v in slots.values():
            found.extend(_stubs(v))
    return found


def check(shots: dict, script_text: str) -> list[str]:
    problems: list[str] = []
    cuts = shots.get("cuts", {})
    scenes = {s["id"]: s for s in shots.get("scenes", [])}
    shot_list = shots.get("shots", [])

    for sid, scene in scenes.items():
        for cut in scene.get("cuts", []):
            if cut not in cuts:
                problems.append(f"scene {sid} names unknown cut {cut}")

    covered = {(s.get("scene"), s.get("cue")) for s in shot_list}
    for sid, cues in script_cues(script_text).items():
        for cue in cues:
            if (sid, cue) not in covered:
                problems.append(f"cue {sid}:{cue} has no shot")

    for s in shot_list:
        if s.get("scene") not in scenes:
            problems.append(f"shot {s.get('id')} names unknown scene {s.get('scene')}")
        tags = set(s.get("tags") or [])
        for stub in _stubs(s.get("slots") or {}):
            if f"stub:{stub}" not in tags:
                problems.append(f"shot {s.get('id')} shows stub {stub} without tag stub:{stub}")

    for cut, spec in cuts.items():
        target = float(spec["target_seconds"])
        total = sum(float(sc.get("duration", 0)) for sc in scenes.values() if cut in sc.get("cuts", []))
        if abs(total - target) > TOLERANCE * target:
            problems.append(f"cut {cut} runs {total:g}s against a {target:g}s target")
    return problems


def main(argv: list[str]) -> int:
    import yaml

    if len(argv) != 3:
        print(__doc__.strip().splitlines()[-1].strip(), file=sys.stderr)
        return 2
    shots = yaml.safe_load(Path(argv[1]).read_text())
    problems = check(shots, Path(argv[2]).read_text())
    for p in problems:
        print(p)
    print(f"{len(problems)} problem(s)")
    return 1 if problems else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

#!/usr/bin/env python3
"""Turn [cue:...] markers in a voiceover script into seconds in the narration audio.

The voiceover is the clock of the whole video. A recognizer returns word timestamps for the audio;
this module lines those words up with the script's words and reads off where each cue landed. The
alignment tolerates a misheard or dropped word: a dropped word gets a start time halfway between the
end of the word before it and the start of the word after it.
"""
from __future__ import annotations

import difflib
import re
from dataclasses import dataclass, field

from verify_shots import CUE_RE, SCENE_RE

WORD_RE = re.compile(r"[a-z0-9']+")


@dataclass
class Scene:
    id: str
    title: str
    text: str = ""
    cues: list[tuple[str, int]] = field(default_factory=list)


def _norm(w: str) -> str:
    return "".join(WORD_RE.findall(w.lower()))


def parse_script(text: str) -> list[Scene]:
    scenes: list[Scene] = []
    words: list[str] = []
    for line in text.splitlines():
        m = SCENE_RE.match(line)
        if m:
            if scenes:
                scenes[-1].text = " ".join(words)
            title = line[m.end():].strip()
            scenes.append(Scene(m.group("id"), title))
            words = []
            continue
        if not scenes or line.lstrip().startswith(">") or not line.strip():
            continue
        for part in re.split(r"(\[cue:[a-z0-9-]+\])", line):
            c = CUE_RE.fullmatch(part)
            if c:
                scenes[-1].cues.append((c.group(1), len(words)))
            else:
                words.extend(part.split())
    if scenes:
        scenes[-1].text = " ".join(words)
    return scenes


def _word_starts(script_words, heard):
    ref = [_norm(w) for w in script_words]
    hyp = [_norm(w) for w, _, _ in heard]
    starts: list[float | None] = [None] * len(ref)
    ends: list[float | None] = [None] * len(ref)
    sm = difflib.SequenceMatcher(a=ref, b=hyp, autojunk=False)
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag in ("equal", "replace") and (i2 - i1) == (j2 - j1):
            for k in range(i2 - i1):
                starts[i1 + k] = heard[j1 + k][1]
                ends[i1 + k] = heard[j1 + k][2]
    for i, s in enumerate(starts):
        if s is None:
            prev_end = next((ends[j] for j in range(i - 1, -1, -1) if ends[j] is not None), 0.0)
            next_start = next((starts[j] for j in range(i + 1, len(starts)) if starts[j] is not None), prev_end)
            starts[i] = (prev_end + next_start) / 2
    return starts, ends


def cue_times(scene: Scene, heard: list[tuple[str, float, float]]) -> dict[str, float]:
    words = scene.text.split()
    starts, ends = _word_starts(words, heard)
    last_end = next((e for e in reversed(ends) if e is not None), heard[-1][2] if heard else 0.0)
    out = {}
    for cue, n in scene.cues:
        out[cue] = round(starts[n], 3) if n < len(words) else round(last_end, 3)
    return out
```

#!/usr/bin/env python3
"""Write timing.json from per-scene narration audio.

Usage:
    python3 timing.py demo/script.md demo/voice demo/timing.json

Reads demo/voice/<scene>.wav for every scene in the script, transcribes it with word timestamps, and
writes {scene: {"duration": seconds, "cues": {cue: seconds}}}. Uses mlx-whisper on Apple silicon
and openai-whisper elsewhere; install one of them in the rig phase.
"""
from __future__ import annotations

import json
import sys
from pathlib import Path

from cues import cue_times, parse_script

MODEL = "mlx-community/whisper-large-v3-turbo"


def transcribe_words(path: Path) -> list[tuple[str, float, float]]:
    try:
        import mlx_whisper
        result = mlx_whisper.transcribe(str(path), path_or_hf_repo=MODEL, word_timestamps=True)
    except ImportError:
        import whisper
        result = whisper.load_model("turbo").transcribe(str(path), word_timestamps=True)
    return [(w["word"], float(w["start"]), float(w["end"]))
            for seg in result["segments"] for w in seg.get("words", [])]


def duration(path: Path) -> float:
    import soundfile as sf
    info = sf.info(str(path))
    return info.frames / info.samplerate


def main(argv: list[str]) -> int:
    script, voice_dir, out = Path(argv[1]), Path(argv[2]), Path(argv[3])
    timing = {}
    for scene in parse_script(script.read_text()):
        wav = voice_dir / f"{scene.id}.wav"
        if not wav.exists():
            print(f"missing {wav}", file=sys.stderr)
            return 1
        timing[scene.id] = {"duration": round(duration(wav), 3),
                            "cues": cue_times(scene, transcribe_words(wav))}
        print(f"{scene.id}: {timing[scene.id]['duration']}s, {len(timing[scene.id]['cues'])} cues")
    out.write_text(json.dumps(timing, indent=1))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

```python
#!/usr/bin/env python3
"""Build the props file Remotion renders a cut from.

Every number the video shows is computed here from three inputs: the shot list, the narration
timing and the demo driver's event log. Nothing is typed into the edit.

Usage:
    python3 build_props.py demo/shots.yaml demo/timing.json demo/events.jsonl short demo/props-short.json
"""
from __future__ import annotations

import json
import math
import sys
from pathlib import Path

WIDTH, HEIGHT = 1920, 1080


def _badges(shot: dict) -> list[str]:
    out = []
    speed = shot.get("speed", 1)
    if speed and speed != 1:
        out.append(f"{speed:g}×")
    for tag in shot.get("tags") or []:
        if tag.startswith("stub:"):
            out.append("stub · " + tag.split(":", 1)[1].replace("-", " "))
        elif tag == "design":
            out.append("design")
    return out


def _facts(events: list[dict], take: str, speed: float, fps: int) -> list[dict]:
    start = next((e["t"] for e in events if e.get("take") == take and e["kind"] == "capture-start"), None)
    if start is None:
        return []
    return [{"atFrame": round((e["t"] - start) / speed * fps), "name": e["name"], "value": e["value"]}
            for e in events if e.get("take") == take and e["kind"] in ("fact", "outcome")]


def _fill_slots(slots: dict, shot: dict, events: list[dict], fps: int) -> dict:
    out = {}
    for name, slot in (slots or {}).items():
        if isinstance(slot, dict) and slot.get("kind") == "facts-panel":
            slot = {**slot, "facts": _facts(events, shot.get("take", ""), float(shot.get("speed", 1)), fps)}
        out[name] = slot
    return out


def build(shots: dict, timing: dict, events: list[dict], cut: str) -> dict:
    fps = int(shots.get("fps", 30))
    scenes = []
    for sc in shots["scenes"]:
        if cut not in sc["cuts"]:
            continue
        t = timing[sc["id"]]
        total = math.ceil(t["duration"] * fps)
        mine = [s for s in shots["shots"] if s["scene"] == sc["id"]]
        starts = [0 if i == 0 else round(t["cues"][s["cue"]] * fps) for i, s in enumerate(mine)]
        placed = []
        for i, s in enumerate(mine):
            end = starts[i + 1] if i + 1 < len(mine) else total
            placed.append({"id": s["id"], "layout": s["layout"], "from": starts[i],
                           "durationInFrames": max(1, end - starts[i]),
                           "slots": _fill_slots(s.get("slots", {}), s, events, fps),
                           "badges": _badges(s), "take": s.get("take"), "speed": s.get("speed", 1)})
        scenes.append({"id": sc["id"], "title": sc["title"], "audio": f"voice/{sc['id']}.wav",
                       "durationInFrames": total, "shots": placed})
    return {"fps": fps, "width": WIDTH, "height": HEIGHT, "cut": cut, "scenes": scenes}


def main(argv: list[str]) -> int:
    import yaml

    shots = yaml.safe_load(Path(argv[1]).read_text())
    timing = json.loads(Path(argv[2]).read_text())
    events = [json.loads(l) for l in Path(argv[3]).read_text().splitlines() if l.strip()]
    props = build(shots, timing, events, argv[4])
    Path(argv[5]).write_text(json.dumps(props, indent=1))
    frames = sum(s["durationInFrames"] for s in props["scenes"])
    print(f"{argv[4]}: {len(props['scenes'])} scenes, {frames / props['fps']:.1f}s -> {argv[5]}")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

```python
#!/usr/bin/env python3
"""Zip an engagement's demo/ folder into a kit, refusing quietly oversized ones.

The kit usually travels by email, so the limit is checked against the uncompressed size: some mail
gateways unpack attachments to scan them, and a kit that is small only when compressed can still be
rejected.

Usage:
    python3 pack.py demo kit.zip [--limit-mb 20]
"""
from __future__ import annotations

import sys
import zipfile
from pathlib import Path

JUNK = {".DS_Store", "Thumbs.db"}
JUNK_DIRS = {"node_modules", "__pycache__", "out", ".git"}


def _files(kit: Path):
    for p in sorted(kit.rglob("*")):
        if p.is_file() and p.name not in JUNK and not (set(p.relative_to(kit).parts) & JUNK_DIRS):
            yield p


def pack(kit: Path, out: Path, limit_mb: float = 20) -> dict:
    files = list(_files(kit))
    sizes = [(f"{kit.name}/{p.relative_to(kit).as_posix()}", p.stat().st_size) for p in files]
    total = sum(s for _, s in sizes)
    largest = sorted(sizes, key=lambda x: -x[1])[:5]
    ok = total <= limit_mb * 1_000_000
    with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED) as z:
        for p, (arc, _) in zip(files, sizes):
            z.write(p, arc)
    return {"ok": ok, "total_mb": round(total / 1_000_000, 2), "zip_mb": round(out.stat().st_size / 1_000_000, 2),
            "files": len(files), "largest": largest}


def main(argv: list[str]) -> int:
    limit = float(argv[argv.index("--limit-mb") + 1]) if "--limit-mb" in argv else 20
    r = pack(Path(argv[1]), Path(argv[2]), limit)
    print(f"{r['files']} files, {r['total_mb']} MB unpacked, {r['zip_mb']} MB zipped -> {argv[2]}")
    if not r["ok"]:
        print(f"over the {limit:g} MB limit; largest files:")
        for name, size in r["largest"]:
            print(f"  {size / 1_000_000:6.2f} MB  {name}")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

**Step 1: `driver.ts`** — a step runner. Each step is `{name, run: () => Promise<StepResult>, pauseMs}`.
It appends an event per request, invocation ID, fact and outcome to `events.jsonl` with `t` measured
from process start, prints each call in a readable form for the terminal recording, and pauses
between steps so a viewer can read. It takes `--take <name>` and `--pretty`.

```ts
// Copy to demo/driver/driver.ts and fill STEPS for the system under demo.
// Run: npx tsx demo/driver/driver.ts --take run-1 --pretty
import {appendFileSync} from 'node:fs';
import {performance} from 'node:perf_hooks';

type Event = Record<string, unknown> & {kind: string};
type StepResult = {invocationId?: string; facts?: Record<string, unknown>; outcome?: {name: string; value: unknown}};
type Step = {name: string; run: () => Promise<StepResult>; pauseMs?: number};

const args = process.argv.slice(2);
const take = args[args.indexOf('--take') + 1] ?? 'run-1';
const pretty = args.includes('--pretty');
const EVENTS = process.env.DEMO_EVENTS ?? 'demo/events.jsonl';
const BASE = process.env.DEMO_BASE_URL ?? 'http://localhost:8080';

const t = () => Math.round(performance.now()) / 1000;
const emit = (e: Event) => appendFileSync(EVENTS, JSON.stringify({t: t(), take, ...e}) + '\n');
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

export async function call(step: string, method: string, path: string, body?: unknown) {
  emit({kind: 'request', step, method, path, body});
  if (pretty) console.log(`\n$ ${method} ${path}${body ? '  ' + JSON.stringify(body) : ''}`);
  const res = await fetch(BASE + path, {method, headers: {'content-type': 'application/json'},
    body: body ? JSON.stringify(body) : undefined});
  const json = await res.json().catch(() => ({}));
  emit({kind: 'response', step, status: res.status, body: json});
  if (pretty) console.log(`  ${res.status}  ${JSON.stringify(json)}`);
  return json as Record<string, unknown>;
}

const STEPS: Step[] = [
  // Example shape. Replace with the real steps in phase 09.
  // {name: 'start', pauseMs: 1500, run: async () => {
  //   const r = await call('start', 'POST', '/journeys/example/run', {facts: {}});
  //   return {invocationId: String(r.invocationId)};
  // }},
];

emit({kind: 'capture-start'});
for (const s of STEPS) {
  const r = await s.run();
  if (r.invocationId) emit({kind: 'invocation', step: s.name, id: r.invocationId});
  for (const [name, value] of Object.entries(r.facts ?? {})) emit({kind: 'fact', step: s.name, name, value});
  if (r.outcome) emit({kind: 'outcome', step: s.name, ...r.outcome});
  await sleep(s.pauseMs ?? 1000);
}
emit({kind: 'capture-end'});
```

//   UI_ROUTE='http://localhost:<UI_PORT>/<route>/{id}' STEP=start npx tsx demo/capture/browser.ts out/center.webm

import {readFileSync, existsSync, renameSync} from 'node:fs';

const out = process.argv[2] ?? 'demo/assets/center.webm';
const route = process.env.UI_ROUTE ?? '';
const step = process.env.STEP ?? '';
const EVENTS = process.env.DEMO_EVENTS ?? 'demo/events.jsonl';
const read = () => (existsSync(EVENTS) ? readFileSync(EVENTS, 'utf8').trim().split('\n').filter(Boolean).map((l) => JSON.parse(l)) : []);

}
await page.waitForTimeout(1500);
const video = page.video();
await context.close();
await browser.close();
if (video) renameSync(await video.path(), out);

  `diff <(jq -c 'del(.t,.id)' run1.jsonl) <(jq -c 'del(.t,.id)' run2.jsonl)` prints nothing.
- **10-capture.md**: one agent per surface, each given the agent kernel: (a) takes for every
  three-pane and editor-terminal shot via `run-take.sh`, one take name per shot list `take`; (b)
  code tokens for every code shot; (c) figures copied to `demo/remotion/public/figures/`; (d) for an
  editor shot, a screen recording of the editor at 1920x1080 with the real compiler error visible,
  or `capture: manual` with a runbook entry. Every asset lands in `demo/remotion/public/` under the
  path its slot names.
- **11-voice.md**: split the script per scene with `python3 <plugin>/scripts/cues.py demo/script.md demo/voice`;
  voice each scene (for `voice-lab`: `python narrate.py demo/voice/<id>.txt --ref <ref.wav> --out demo/remotion/public/voice/<id>.wav`);
  add failed words to the engine's lexicon and re-voice; then
  `python3 <plugin>/scripts/timing.py demo/script.md demo/remotion/public/voice demo/timing.json`.
- **12-compose.md**: per cut,
  `python3 <plugin>/scripts/build_props.py demo/shots.yaml demo/timing.json demo/events.jsonl <cut> demo/props-<cut>.json`,
  preview with `npm run studio`, render with
  `npx remotion render src/index.ts Cut out/<cut>.mp4 --props=../props-<cut>.json` from `demo/remotion`.
  A layout the narrative needs and the starter lacks is added to the copy in `demo/remotion`, then
  proposed back to the plugin.
- **13-review.md**: the checklist, run against the draft before the owner sees it: every cut follows
  a sentence; no text under 22 px at 1080p; every speed-up shows its factor; every stub and design
  label present; every number on screen traceable to `events.jsonl`; the climax evidence (for
  example the timer) is real elapsed time; runtime within 10% of target. Then the owner's notes,
  applied and re-rendered.
- **14-deliver.md**: final renders per cut, an SRT per cut built from `script.md` and
  `timing.json` (one caption per sentence), a thumbnail per cut
  (`npx remotion still src/index.ts Cut out/<cut>.png --frame=<climax frame> --props=...`), and a
  hand-back note listing every `capture: manual` shot and every disagreement ruled in phase 07.
- **ledger.md**: copy arch-docs' ledger rules; format adds `machine: author | target` and the kit's
  version. `phase` is one of the 14 phase names.
- **agent-kernel.md**: arch-docs' ten points (mode line; absolute working directory; numbered
  read-first list; one source of truth with override order; ownership and forbidden files; a loop
  with the actual commands; definition of done by name; invention guard; a voice line; a capped
  report asking what went wrong), plus an eleventh: every agent that shows a number on screen
  states which `events.jsonl` line it came from.
- **failure-modes.md**: starts with the design's failure-handling list, each as "what happened, what
  it cost, the rule that prevents it". Engagements append to it.

**Step 1:** Write the eleven files. **Step 2:** Lint them as in Task 9. **Step 3:** Commit:
`git commit -m "feat(demo-reel): target-side phase references, ledger and agent kernel"`.
